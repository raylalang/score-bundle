\begin{frame}{Where the thesis stands}
  \textbf{What it answers, three times.} Given a score and an observation,
  the expression of every note is recovered with calibrated uncertainty:
  from aligned MIDI (piano), from tracked audio (strings and winds) and
  from the waveform itself. Each step was tested on held-out pieces.
  \vspace{2mm}

  \textbf{What it found about the structure.} The score graph's measured
  value is smoothing in score time over about two beats. The pitch term
  adds nothing on piano, on strings and winds, or in the waveform, at
  chain, phrase or track scale.
  \vspace{2mm}

  \textbf{What it cannot yet say.}
  \begin{itemize}
    \item which \emph{score structure beyond time} performance follows,
      which is the question the title implies;
    \item what the recovered representation is \emph{for}, beyond the
      downstream tasks of Chapter 6.
  \end{itemize}
  \vspace{2mm}
  The proposal closes the first gap and gives the second a concrete form.
\end{frame}

\begin{frame}{The deferred alternative: an across-performer model}
  \small
  The feedback application needs a band that means \emph{the range of
  reasonable interpretations}, not \emph{what this one performer would
  have done}. That band comes from a model of performer variation. For
  a piece with performances $r = 1, \dots, R$,
  \[
    y^{(r)} = \mu + \delta^{(r)}, \qquad
    \mu \sim \N\bigl(0,\, B_\mu \otimes K_{\mathcal G}(s)\bigr), \qquad
    \delta^{(r)} \overset{\text{iid}}{\sim} \N\bigl(0,\, B_\delta \otimes K_{\mathcal G}(s)\bigr),
  \]
  so that the shared mean $\mu$ and every deviation $\delta^{(r)}$ live
  on the same score graph, and the feature kernels and noise of the
  proposed model are kept. A new performance $y^{(\star)}$ is then scored
  against
  \[
    p\bigl(y^{(\star)}_i \mid y^{(1)}, \dots, y^{(R)}\bigr),
    \qquad
    \operatorname{Var} = \underbrace{\operatorname{Var}[\mu_i \mid \cdot]}_{\text{uncertainty about the shared mean}}
      + \underbrace{(B_\delta)_{cc}\,[K_{\mathcal G}]_{ii}}_{\text{performer variation}}
      + \varsigma_c^2,
  \]
  which is the band a feedback view should draw. ASAP supplies the data:
  27 performances of the 21st sonata's first movement, 21 of the 31st, 17
  of the 23rd.
  \vspace{2mm}

  \textbf{Why deferred.} It is a new model with its own fitting, its own
  identifiability question ($\mu$ against $\delta$ when $R$ is small) and
  its own evaluation. One to two weeks. It is the right Future Work item
  for the application, and the motif study does not depend on it.
\end{frame}

\begin{frame}{The bridge: from annotated occurrences to an edge family}
  \footnotesize
  \textbf{What the annotations give.} Occurrences $O_1, \dots, O_K$ as
  note sets with a span $[b_k^{\text{start}}, b_k^{\text{end}}]$ in beats
  and a type $q(k)$. Returns of one motif differ in length, so a
  note-to-note correspondence is not well defined in general.
  \vspace{1mm}

  \textbf{Correspondence by relative position, many to many.} Each note
  sits at its relative position inside its occurrence,
  \[
    u_i = \frac{b_i - b_k^{\text{start}}}{\,b_k^{\text{end}} - b_k^{\text{start}}\,} \in [0, 1],
    \qquad i \in O_k,
  \]
  and connect notes of two same-type occurrences by closeness in that
  coordinate, with the Gaussian form the time edges already use,
  \[
    \bigl(W_{\text{motif}}\bigr)_{ij} \;=\;
    \sum_{k \neq l,\; q(k) = q(l)} \mathbf 1[i \in O_k]\, \mathbf 1[j \in O_l]\;
    \exp\!\Bigl(-\frac{(u_i - u_j)^2}{2\,\ell_u^2}\Bigr),
  \]
  with $\ell_u$ fixed at about a tenth of the span. Equal-length returns
  recover one-to-one edges; unequal ones get a soft many-to-many block
  that needs no alignment. Nothing is learned here.
  \vspace{1mm}

  \textbf{The score graph.} The thesis's adjacency gains the family with
  one learned weight,
  \[
    W \;=\; W_{\text{time}} + w_{\text{m}}\, W_{\text{motif}}, \qquad
    \Lg = D - W, \qquad
    K_{\mathcal G}(s) = U\,\diag\bigl(g(\nu; s)\bigr)\,U^{\top},
  \]
  and the rest of the model, $K = B \otimes K_{\mathcal G}(s) + \sum_f \diag(c_f) \otimes X_f X_f^{\top} + \diag(\varsigma^2) \otimes I$,
  is unchanged. $w_{\text{m}}$ joins $\theta$ in the per-piece evidence,
  which can switch the family off: the falsification route.
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{The GP side: what a motif edge does to a hidden note}
  \footnotesize
  In the nested single-channel case with the additive kernel,
  $g(\nu; s) = 1/(1 + s\nu)$, the prior is the Gaussian Markov random
  field with precision $Q = I + s\Lg$, and the conditional of a hidden
  note $h$ given the other notes is explicit:
  \[
    \mathbb E\bigl[y_h \mid y_{-h}\bigr] \;=\; \frac{s \sum_{j} W_{hj}\, y_j}{1 + s\, d_h},
    \qquad
    \operatorname{Var}\bigl[y_h \mid y_{-h}\bigr] \;=\; \frac{1}{1 + s\, d_h},
    \qquad d_h = \sum_j W_{hj}.
  \]
  Splitting the degree into the two families, $d_h = d_h^{\text{time}} + w_{\text{m}} d_h^{\text{motif}}$,
  \[
    \mathbb E\bigl[y_h \mid y_{-h}\bigr] \;=\;
    \frac{s\, d_h^{\text{time}}}{1 + s d_h}\;\overline{y}^{\,\text{time}}_h
    \;+\; \frac{s\, w_{\text{m}}\, d_h^{\text{motif}}}{1 + s d_h}\;\overline{y}^{\,\text{motif}}_h,
  \]
  the degree-weighted mean of the temporal neighbours and the
  relative-position-weighted mean of the other occurrences' notes. The posterior of the
  full model, $m = C_{\cdot o} K_{oo}^{-1} \tilde y_o$, mixes the same two
  sources through the coupling $B$, the feature kernels and the noise, but
  this is the mechanism.
  \vspace{1mm}

  \textbf{Two readable predictions.} If performers shape a motif the same
  way at each return, $\overline{y}^{\,\text{motif}}_h$ is closer to the
  truth than $\overline{y}^{\,\text{time}}_h$ and the evidence raises
  $w_{\text{m}}$: the recovery error on hidden motif notes falls (H1,
  RMSE) and the band tightens by exactly the extra degree, which is the
  calibration test (H1, NLL and coverage). If they do not, the counterpart
  mean is noise, the evidence drives $w_{\text{m}}$ toward zero, and the
  arms coincide with the time-only graph: the falsification outcome.
\end{frame}

\begin{frame}{Methodology: arms, masking, data, hypotheses}
  \small
  \textbf{Arms}, paired per piece and seed, the per-piece bias column in
  every arm: time-only graph (the attribution winner), time $+$ pitch
  (the current model), time $+$ annotated motif edges. Predicted edges
  are a second stage, not part of this study.
  \vspace{1mm}

  \textbf{Masking.} Hide notes inside motif occurrences while at least
  one other occurrence of the same type stays fully observed, at the
  usual 40 percent rate on the remaining notes. Score timing, articulation and velocity by
  paired RMSE and NLL, and coverage of the 90 percent band, on the hidden
  motif notes and on all hidden notes.
  \vspace{1mm}

  \textbf{Data.} BPS-motif annotates the first movements of all 32
  Beethoven sonatas. ASAP holds 25 of them, over 200 performances, nine
  pieces already inside the thesis's pools. The 400-note excerpts are the
  expositions, where the motif returns.
  \vspace{1mm}

  \textbf{Hypotheses, fixed before the first run.} H1: on hidden motif
  notes, the annotated-motif arm improves timing and velocity NLL against
  the time-only graph, with an interval excluding zero. H1b: it also
  improves RMSE there. Report-only: the learned $w_{\text{m}}$ per piece
  and the effect on non-motif notes. Falsification: an H1 interval that
  includes zero means the structure performance follows is temporal
  smoothing, and the thesis says so.
\end{frame}

\begin{frame}{Stage two, later: occurrences predicted by the ISMIR 2026 model}
  \footnotesize
  \textbf{Typed note graph and note states.} Notes are vertices, with edge
  types for simultaneity, temporal proximity and overlap. A pretrained
  encoder gives $\mathbf h_i^{(0)}$, and relational graph convolutions
  update
  \[
    \mathbf h_i^{(\ell+1)} = \phi\Bigl( W^{(\ell)}_{\text{self}} \mathbf h_i^{(\ell)}
      + \sum_{r} \frac{W^{(\ell)}_r}{|\mathcal N_r(i)|} \sum_{j \in \mathcal N_r(i)} \mathbf h_j^{(\ell)} \Bigr),
    \qquad \mathbf h_i \in \R^{d} \text{ after } L \text{ layers.}
  \]
  \textbf{Two heads.} A motif-note probability and a pair affinity on
  candidate pairs $\mathcal C$ taken from the graph edges,
  \[
    m_i = \sigma\bigl(f_{\text{MNID}}(\mathbf h_i)\bigr), \qquad
    \hat a_{ij} = \sigma\bigl(\mathrm{MLP}(\mathbf g_{ij})\bigr), \quad
    \mathbf g_{ij} = \bigl[\mathbf h_i, \mathbf h_j, \mathbf h_i \odot \mathbf h_j,
      \tfrac{|t_i - t_j|}{q + \varepsilon}, \tfrac{|p_i - p_j|}{s_p}, \mathbf 1[t_i = t_j]\bigr].
  \]
  \textbf{Training by label propagation.} Annotated occurrences $O_k$ seed
  $S_{ki} = 1/|O_k|$ on their notes, and memberships diffuse along the
  learned affinities with restart,
  \[
    T = D^{-1}(A + s_{\text{self}} I), \qquad
    M^{(t+1)} = \alpha S + (1 - \alpha)\, M^{(t)} T, \qquad
    t_{\max} = 4,\; \alpha = 0.55,
  \]
  with losses at the note-pair, occurrence and motif-type levels on $M$.
  \textbf{Decoding.} The gated affinity $A^{\text{dec}}_{ij} = A_{ij}(m_i m_j)^{\alpha_{\text{dec}}/2}$
  is thresholded at $\tau_{\text{occ}}$, connected components become
  occurrences $\hat O_1, \dots, \hat O_{\hat K}$, and a Gaussian mixture
  on pooled states $\hat{\mathbf z}_k$ assigns each occurrence a type
  $q(k)$.
  \vspace{1mm}

  \textbf{What stage two takes:} the occurrences $\hat O_k$ with spans
  and types $q(k)$ in place of the annotations, on any score
  (establishment F1 $0.732$ on BPS-motif). It waits for stage one's answer.
\end{frame}
