\begin{frame}{Where the thesis stands}
  \textbf{What it answers, three times}
  \begin{itemize}
    \item score $+$ observation $\to$ the expression of every note, with calibrated uncertainty
    \item from aligned MIDI (piano), tracked audio (strings and winds), the waveform itself
    \item each step tested on held-out pieces
  \end{itemize}
  \vspace{1mm}
  \textbf{What it found about the structure}
  \begin{itemize}
    \item the score graph's measured value: smoothing in score time over about two beats
    \item the pitch term adds nothing, on either corpus, at chain, phrase or track scale
  \end{itemize}
  \vspace{1mm}
  \textbf{What it cannot yet say}
  \begin{itemize}
    \item which \emph{score structure beyond time} performance follows (the title's question)
    \item what the recovered representation is \emph{for}, beyond the downstream tasks of Chapter 6
  \end{itemize}
  \vspace{1mm}
  The proposal closes the first gap and gives the second a concrete form.
\end{frame}

\begin{frame}{The deferred alternative: an across-performer model}
  \footnotesize
  \begin{itemize}
    \item a feedback view needs a band meaning \emph{the range of reasonable interpretations}; that is a model of performer variation, performances $r = 1, \dots, R$ of one piece:
  \end{itemize}
  \vspace{-2mm}
  \[
    y^{(r)} = \mu + \delta^{(r)}, \qquad
    \mu \sim \N\bigl(0,\, B_\mu \otimes K_{\mathcal G}(s)\bigr), \qquad
    \delta^{(r)} \overset{\text{iid}}{\sim} \N\bigl(0,\, B_\delta \otimes K_{\mathcal G}(s)\bigr),
  \]
  \vspace{-1mm}
  \begin{itemize}
    \item shared mean $\mu$ and every deviation $\delta^{(r)}$ on the same score graph; feature kernels and noise kept
    \item a new performance $y^{(\star)}$ is scored against
  \end{itemize}
  \vspace{-2mm}
  \[
    p\bigl(y^{(\star)}_i \mid y^{(1)}, \dots, y^{(R)}\bigr),
    \qquad
    \operatorname{Var} = \underbrace{\operatorname{Var}[\mu_i \mid \cdot]}_{\text{uncertainty about the shared mean}}
      + \underbrace{(B_\delta)_{cc}\,[K_{\mathcal G}]_{ii}}_{\text{performer variation}}
      + \varsigma_c^2,
  \]
  \vspace{-1mm}
  \begin{itemize}
    \item data: ASAP, 27 performances of the 21st sonata's first movement, 21 of the 31st, 17 of the 23rd
    \item \textbf{deferred}: a new model with its own fitting, identifiability ($\mu$ against $\delta$ at small $R$) and evaluation, one to two weeks; the Future Work item for the application
  \end{itemize}
\end{frame}

\begin{frame}{The bridge: from annotated occurrences to an edge family}
  \footnotesize
  \begin{itemize}
    \item the annotations give occurrences $O_1, \dots, O_K$: note sets with a span $[b_k^{\text{start}}, b_k^{\text{end}}]$ in beats and a type $q(k)$
    \item returns of one motif differ in length, so no note-to-note map in general
    \item instead, relative position inside the occurrence, and closeness in that coordinate, many to many:
  \end{itemize}
  \vspace{-2mm}
  \[
    u_i = \frac{b_i - b_k^{\text{start}}}{\,b_k^{\text{end}} - b_k^{\text{start}}\,} \in [0, 1],
    \qquad i \in O_k,
  \]
  \vspace{-3mm}
  \[
    \bigl(W_{\text{motif}}\bigr)_{ij} \;=\;
    \sum_{k \neq l,\; q(k) = q(l)} \mathbf 1[i \in O_k]\, \mathbf 1[j \in O_l]\;
    \exp\!\Bigl(-\frac{(u_i - u_j)^2}{2\,\ell_u^2}\Bigr),
  \]
  \vspace{-1mm}
  \begin{itemize}
    \item $\ell_u$ fixed at a tenth of the span; equal-length returns give one-to-one edges, unequal ones a soft block, no alignment, nothing learned
    \item the thesis's adjacency gains the family with one learned weight:
  \end{itemize}
  \vspace{-2mm}
  \[
    W \;=\; W_{\text{time}} + w_{\text{m}}\, W_{\text{motif}}, \qquad
    \Lg = D - W, \qquad
    K_{\mathcal G}(s) = U\,\diag\bigl(g(\nu; s)\bigr)\,U^{\top},
  \]
  \vspace{-1mm}
  \begin{itemize}
    \item the rest, $K = B \otimes K_{\mathcal G}(s) + \sum_f \diag(c_f) \otimes X_f X_f^{\top} + \diag(\varsigma^2) \otimes I$, is unchanged
    \item $w_{\text{m}}$ joins $\theta$ in the per-piece evidence, which can switch the family off: the falsification route
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{The GP side: what a motif edge does to a hidden note}
  \footnotesize
  \begin{itemize}
    \item single channel, additive kernel $g(\nu; s) = 1/(1 + s\nu)$: the prior is the GMRF with precision $Q = I + s\Lg$
    \item the conditional of a hidden note $h$ given the others is explicit:
  \end{itemize}
  \vspace{-2mm}
  \[
    \mathbb E\bigl[y_h \mid y_{-h}\bigr] \;=\; \frac{s \sum_{j} W_{hj}\, y_j}{1 + s\, d_h},
    \qquad
    \operatorname{Var}\bigl[y_h \mid y_{-h}\bigr] \;=\; \frac{1}{1 + s\, d_h},
    \qquad d_h = \sum_j W_{hj}.
  \]
  \vspace{-1mm}
  \begin{itemize}
    \item split the degree into the two families, $d_h = d_h^{\text{time}} + w_{\text{m}} d_h^{\text{motif}}$:
  \end{itemize}
  \vspace{-2mm}
  \[
    \mathbb E\bigl[y_h \mid y_{-h}\bigr] \;=\;
    \frac{s\, d_h^{\text{time}}}{1 + s d_h}\;\overline{y}^{\,\text{time}}_h
    \;+\; \frac{s\, w_{\text{m}}\, d_h^{\text{motif}}}{1 + s d_h}\;\overline{y}^{\,\text{motif}}_h,
  \]
  \vspace{-1mm}
  \begin{itemize}
    \item a mean of the temporal neighbours plus a mean of the other occurrences' notes, weighted by relative position
    \item the full posterior $m = C_{\cdot o} K_{oo}^{-1} \tilde y_o$ mixes the same two sources through $B$, the feature kernels and the noise
  \end{itemize}
  \vspace{1mm}
  \textbf{Two readable predictions}
  \begin{itemize}
    \item performers shape a motif the same way at each return: $\overline{y}^{\,\text{motif}}_h$ is closer than $\overline{y}^{\,\text{time}}_h$, the evidence raises $w_{\text{m}}$, error falls (H1b) and the band tightens by exactly the extra degree (H1)
    \item they do not: the counterpart mean is noise, the evidence drives $w_{\text{m}}$ to zero, the arms coincide with the time-only graph (falsification)
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Methodology: arms, masking, data, hypotheses}
  \footnotesize
  \textbf{Arms}, paired per piece and seed, a per-piece bias column in every arm
  \begin{itemize}
    \item time-only graph (the attribution winner); time $+$ pitch (the current model); time $+$ annotated motif edges
    \item predicted edges are a second stage, not part of this study
  \end{itemize}
  \textbf{Masking and scoring}
  \begin{itemize}
    \item hide whole occurrences while another occurrence of the type stays observed; also the usual 40 percent at random
    \item timing, articulation, velocity: paired RMSE and NLL, coverage of the 90 percent band, on hidden motif notes
  \end{itemize}
  \textbf{Data}
  \begin{itemize}
    \item BPS-motif annotates all 32 Beethoven sonata first movements; ASAP holds 25; 400-note excerpts, the expositions; confirmation pieces excluded
  \end{itemize}
  \textbf{Hypotheses, fixed before the first run}
  \begin{itemize}
    \item H1: on hidden motif notes, the motif arm lowers timing NLL against the time-only graph, interval excluding zero; H1b: RMSE too
    \item report-only: $w_{\text{m}}$ per piece, the other channels, the random scheme; falsification: an H1 interval covering zero means the structure is temporal smoothing, and the thesis says so
  \end{itemize}
\end{frame}

\begin{frame}{Stage two: occurrences predicted by the ISMIR 2026 model (measured 17 November, development, no claim)}
  \scriptsize
  \begin{itemize}
    \item typed note graph: notes are vertices, edge types for simultaneity, temporal proximity, overlap; a pretrained encoder gives $\mathbf h_i^{(0)}$, relational graph convolutions update
  \end{itemize}
  \[
    \mathbf h_i^{(\ell+1)} = \phi\Bigl( W^{(\ell)}_{\text{self}} \mathbf h_i^{(\ell)}
      + \sum_{r} \frac{W^{(\ell)}_r}{|\mathcal N_r(i)|} \sum_{j \in \mathcal N_r(i)} \mathbf h_j^{(\ell)} \Bigr),
    \qquad \mathbf h_i \in \R^{d} \text{ after } L \text{ layers.}
  \]
  \vspace{-1mm}
  \begin{itemize}
    \item two heads: a motif-note probability and a pair affinity on candidate pairs $\mathcal C$ from the graph edges
  \end{itemize}
  \[
    m_i = \sigma\bigl(f_{\text{MNID}}(\mathbf h_i)\bigr), \qquad
    \hat a_{ij} = \sigma\bigl(\mathrm{MLP}(\mathbf g_{ij})\bigr), \quad
    \mathbf g_{ij} = \bigl[\mathbf h_i, \mathbf h_j, \mathbf h_i \odot \mathbf h_j,
      \tfrac{|t_i - t_j|}{q + \varepsilon}, \tfrac{|p_i - p_j|}{s_p}, \mathbf 1[t_i = t_j]\bigr].
  \]
  \vspace{-1mm}
  \begin{itemize}
    \item training by label propagation: annotated occurrences seed $S_{ki} = 1/|O_k|$, memberships diffuse along the learned affinities with restart
  \end{itemize}
  \[
    T = D^{-1}(A + s_{\text{self}} I), \qquad
    M^{(t+1)} = \alpha S + (1 - \alpha)\, M^{(t)} T, \qquad
    t_{\max} = 4,\; \alpha = 0.55,
  \]
  \vspace{-1mm}
  \begin{itemize}
    \item losses at the note-pair, occurrence and motif-type levels on $M$
    \item decoding: gated affinity $A^{\text{dec}}_{ij} = A_{ij}(m_i m_j)^{\alpha_{\text{dec}}/2}$ thresholded at $\tau_{\text{occ}}$, connected components are occurrences $\hat O_k$, a Gaussian mixture on pooled states gives each a type $q(k)$
    \item \textbf{measured 17 November}: held-out fold predictions ($\tau_{\text{occ}}$ quantile $0.86$, F1 median $0.78$) replace the annotations as edge source; shared form: timing log score $-0.034$ $[-0.085,\,+0.003]$, $0.48$ of $-0.071$; articulation RMSE $-0.0135$ $[-0.032,\,+0.003]$, $0.4$ of $-0.0345$; per-piece correlation $0.84$. H3 named the per-channel form: $+0.036$ $[-0.010,\,+0.095]$, not supported
  \end{itemize}
\end{frame}
