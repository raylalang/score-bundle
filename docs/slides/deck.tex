% ============================================== PART I  THE THESIS

\begin{frame}{Expressive performance}
  The score fixes pitches and note values. Performers add the rest:
  when, how long, how loud, and on strings and winds, how the pitch
  moves inside the note. These deviations are the expression.
  \vspace{-2mm}
  \begin{center}
  \includegraphics[height=0.66\textheight]{deck_concept.png}
  \end{center}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{The thesis, and the three problems}
  \textbf{Thesis: one Gaussian process on the score graph turns a known
  score and a performance into per-note expressive variables, each with
  an error bar you can trust.}
  \vspace{2mm}

  Three observation problems:
  \begin{enumerate}
    \item \textbf{Piano}: timing, articulation, velocity, read directly
          from aligned MIDI. Are the error bars calibrated, note by
          note?
    \item \textbf{Strings and winds}: six variables, now with
          intonation and vibrato, \textbf{estimated} from audio,
          noisily. Does the same prior still work?
    \item \textbf{Audio directly}: no pitch tracker at all. Can the
          waveform itself be the observation?
  \end{enumerate}
\end{frame}

\begin{frame}{Related work}
  \begin{columns}[T,onlytextwidth]
  \column{0.50\textwidth}
  \textbf{Expressive performance modelling}
  {\small
  \begin{itemize}
    \item rule systems: KTH performance rules (Friberg et al.\ 2006)
    \item linear basis-function models (Grachten and Widmer 2012)
    \item deep rendering models: VirtuosoNet (Jeong et al.\ 2019)
  \end{itemize}
  All output one rendering or corpus statistics, none a calibrated
  per-note posterior.}
  \column{0.48\textwidth}
  \textbf{Gaussian processes and graphs}
  {\small
  \begin{itemize}
    \item GPs over graphs (Venkitaraman et al.\ 2018)
    \item graph Mat\'ern kernels (Borovitskiy et al.\ 2021)
    \item multi-output coupling, ICM (Bonilla et al.\ 2008)
    \item differentiable synthesis (Engel et al.\ 2020)
  \end{itemize}}
  \end{columns}
  \vfill
  This work combines them: a graph GP over score positions, with
  per-note side information, for calibrated performance inference, down
  to the waveform.
\end{frame}

\divider{The model}{A multi-output Gaussian process on the score graph}

% ---------------------------------------------------------------------
\begin{frame}{Approach: one prior, three kinds of observation}
  One \textbf{Gaussian process (GP)} prior over per-note expressive fields, on
  a graph built from the score. Only the observation changes: per-note
  MIDI values, noisy $f_0$-derived targets, or the waveform itself. A
  pretrained music model supplies input features to all.
  \figslotw{deck_programme.png}{0.9}{the shared prior with three swappable
    likelihood blocks (MIDI / $f_0$ / waveform)}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{The model (1): the score graph}
  Nodes are the $N$ written notes: pitch $p_i$, beat onset $b_i$, beat
  duration $d_i$. Edges connect notes close in score time and pitch
  (neighbours should be played similarly):
  \vspace{-1mm}
  \begin{equation*}
    W_{ij} \;=\; \exp\!\Bigl( -\tfrac{(b_i - b_j)^2}{2\,\ell_b^2}
                             \;-\; \tfrac{(p_i - p_j)^2}{2\,\ell_p^2} \Bigr),
    \qquad
    \Lg \;=\; D - W .
  \end{equation*}
  \vspace{-2mm}
  {\centering\small $\ell_b, \ell_p$: length-scales. $D_{ii} = \sum_j W_{ij}$.
  $\Lg$: the graph Laplacian. $W, D, \Lg \in \R^{N \times N}$.\par}
  \figslotw{deck_scoregraph.png}{0.80}{a few bars of real score as nodes,
    edges drawn by time / pitch / chord relation}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{The model (2): a GP on the graph, channels coupled}
  Eigendecompose the Laplacian, $\Lg = U \diag(\nu)\, U^{\!\top}$
  ($U \in \R^{N \times N}$ eigenvectors, $\nu \in \R^N$ eigenvalues), and
  shape its spectrum with one parameter $s$:
  \begin{equation*}
    K_G(s) \;=\; U\, g(\nu; s)\, U^{\!\top},
    \qquad
    g(\nu; s) \;=\; \frac{1}{1 + s\,\nu}
    \quad\text{(graph Mat\'ern, } \alpha = 1\text{)} .
  \end{equation*}
  Smooth-on-the-graph components keep their variance, rough ones are shrunk.
  \vspace{1mm}

  The channels are correlated, so they share the graph through a learned
  coupling matrix $B$ ($c, c'$ index the channels):
  \begin{equation*}
    \operatorname{Cov}\bigl[\, y_{i,c},\; y_{j,c'} \bigr]
      \;=\; B_{cc'}\, \bigl[K_G(s)\bigr]_{ij}
    \qquad\Longleftrightarrow\qquad
    \operatorname{Cov}[\,y\,] \;=\; B \otimes K_G(s).
  \end{equation*}
  An observed channel at one note then informs the other channels at
  neighbouring notes.
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{The model (3): side information as kernels}
  Any per-note side information enters the same way. Two sets, each one
  row per note ($N \times d$): score descriptors $X_{\mathrm{feat}}$
  (register, duration, meter, position, and related) and per-note
  features $H$ from a small music model trained on MAESTRO.
  \vspace{1mm}

  A linear mean with unknown weights $\beta$, integrated out, becomes a
  kernel term ($X$ = either feature matrix, $f_G$ = the graph GP of
  the previous slide, $c_X$ = the prior scale of set $X$'s weights):
  \begin{equation*}
    y \;=\; X\beta + f_G, \quad \beta \sim \N(0,\, c_X\,I)
    \qquad\Longrightarrow\qquad
    \operatorname{Cov}[\,y\,] \;=\; c_X\, X X^{\!\top} + \operatorname{Cov}[\,f_G\,].
  \end{equation*}
  \begin{itemize}
    \item So the regression weights are not fixed once across the corpus: they
          are \textbf{re-inferred for every piece}, jointly with everything
          else, and their uncertainty enters the posterior.
    \item This per-piece adaptation turns out to be the main source of the
          accuracy gain (results).
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{The model (4): the covariance of one performance}
  Stack all channels of the $N$ notes into one vector $y$.
  Every component becomes one term of its covariance $K$, built per piece:
  \begin{equation*}
    K \;=\;
    \underbrace{\textcolor{okblue}{B \otimes K_G(s)}}_{\text{coupled graph GP}}
    \;+\; \underbrace{\textcolor{okamber}{\textstyle\sum_f \diag(c_f)\otimes X_f X_f^{\!\top}}}_{\text{side information}}
    \;+\; \underbrace{\textcolor{okverm}{\diag(\varsigma^2)\otimes I}}_{\text{noise}}
  \end{equation*}
  {\small $f$ runs over the two feature sets, giving per-channel scales
  $c_{\mathrm{feat}}, c_{\mathrm{mm}}$. $\varsigma^2$ = the per-channel
  noise variances.}
  \vspace{1mm}

  All parameters $\theta = (s, B, \{c_f\}, \varsigma^2)$, 16 numbers on
  piano (the construction is identical for any channel count), from one
  exact evidence per piece. Prediction is conjugate:
  \begin{equation*}
    m = C_{\cdot o} K_{oo}^{-1} \tilde y_o, \qquad
    \Sigma_y = C - C_{\cdot o} K_{oo}^{-1} C_{o\cdot},
  \end{equation*}
  {\small $\tilde y_o$: the observed values. $o$: the observed entries of
  $y$. $C$: the noise-free part of $K$. $m, \Sigma_y$: posterior mean and
  covariance. No variational approximation anywhere, and the baselines are
  exact special cases (diagonal $B$, zero $c_f$, fixed cross-piece mean).}
\end{frame}

\begin{frame}{Setup: audio instead of MIDI}
  \begin{itemize}
    \item Data: \textbf{URMP}, 44 chamber pieces (duets to quintets, 13
          instrument types), each instrument recorded on its own
          monophonic track with note annotations and the score. Split by
          composition, since arrangements share recordings: development
          31 pieces (78 tracks, about 17{,}000 notes), one-shot test 13
          pieces.
    \item Pitch now carries expression: players control intonation and
          vibrato within every note.
    \item A pitch tracker (pYIN) turns each track into frames: time,
          $f_0$, voicing confidence. The variables must be
          \textbf{estimated} from these frames, noisily, with a variance
          per note.
    \item Two measurements of every note exist: our tracked curve, and
          the corpus's own annotation of the same track (cleaner, but
          still a measurement). Evaluation uses the second.
  \end{itemize}
\end{frame}

\begin{frame}{From audio to the cents curve}
  A pitch tracker turns each track into frames $(t_j, f_{0,j}, \pi_j)$:
  time, fundamental frequency, voicing confidence.
  \begin{itemize}
    \item \textbf{Frame rule}: note $i$ keeps
          $J_i = \{\, j:\ \text{voiced},\ \pi_j \ge q_{0.2},\
          t_j \in [o_i,\, o_i + u_i) \,\}$, at least 4 frames
          ($o_i, u_i$: the note's annotated onset and duration,
          $q_{0.2}$: the track's lowest confidence quintile).
    \item \textbf{Cents curve}: each kept frame becomes a deviation from
          equal temperament at 440 Hz,
          $x_j = 1200 \log_2 (f_{0,j}/440) - 100\,(p_i - 69)$
          ($p_i$: MIDI pitch).
  \end{itemize}
  \vfill
  The pipeline consumes the tracked curve.
\end{frame}

\begin{frame}{The pitch-curve channels, defined}

% ---------------------------------------------------------------------
\begin{frame}{Estimating the channels with a Gaussian process}
  The note's cents curve as a \textbf{Gaussian process}, two components:
  a damped cosine at the vibrato rate, plus a slow drift (a special case
  of the spectral mixture kernel, Wilson and Adams 2013):
  \begin{equation*}
    k(\Delta t) \;=\;
    \underbrace{w_1\, e^{-2\pi^2 v_1 \Delta t^2} \cos(2\pi \mu_1 \Delta t)}_{\text{vibrato component}}
    \;+\;
    \underbrace{w_2\, e^{-2\pi^2 v_2 \Delta t^2}}_{\text{drift component}}
  \end{equation*}
  {\small $\Delta t$: time lag between two frames (s), so $\tau$ stays
  the timing channel. $\mu_1$: rate (Hz). $w_1, w_2$: powers (cents$^2$).
  $v_1, v_2$: bandwidths (Hz$^2$).}
  \begin{itemize}
    \item The prior states only: the curve's energy sits in a band of
          width $\sqrt{v_1}$ around the rate $\mu_1$, plus a
          low-frequency drift band around zero. Nothing else is assumed.
    \item Fit per note by exact marginal likelihood, variances from
          the evidence curvature.
    \item Channels from the posterior: $c$ the realized average,
          $\gamma$ from the vibrato component, $f = \mu_1$.
  \end{itemize}
\end{frame}

\divider{The waveform as the observation}{Methodology and first development studies, no claims}

% ---------------------------------------------------------------------
\begin{frame}{The method, and where it comes from}
  \begin{columns}[T,onlytextwidth]
  \column{0.52\textwidth}
  \textbf{Reference works}
  {\small
  \begin{itemize}
    \item \textbf{DDSP} (Engel et al.\ 2020): gradients through a
          differentiable synthesizer; MIDI-DDSP (Wu et al.\ 2022) the
          closest cousin for performance control.
    \item GPs for music audio (Alvarado and Stowell 2016): per-note
          windows over a shared time axis.
    \item Spectral mixture kernels (Wilson and Adams 2013): the curve
          prior's two-component kernel is a special case.
  \end{itemize}}
  \column{0.44\textwidth}
  \textbf{What this thesis does with them}
  {\small
  \begin{itemize}
    \item \textbf{Takes}: gradient descent through the synthesizer
          (the implementation idea).
    \item \textbf{Keeps}: the exact amplitude-marginalized likelihood
          and the GP priors --- no neural encoder, no spectral
          surrogate loss as the objective.
    \item \textbf{Measured}: a learned many-component spectrum adds
          nothing over the hand structure on these curves.
  \end{itemize}}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Setup: the data and the signal model}
  \small
  Data: URMP development tracks, audio only. 7 tracks, one per
  instrument, notes cut at annotated boundaries, 16\,kHz. 376 notes in
  the curve studies; 47 chains of 3 to 5 notes (154 boundaries) in the
  boundary study.
  \vspace{3mm}

  A sustained note is a stack of $K$ harmonics. Its audio is
  \begin{equation*}
    x \;=\; \Phi(z)\,a \;+\; \varepsilon,
    \qquad x \in \mathbb R^{m},\quad \Phi(z) \in \mathbb R^{m\times p},\quad
    a \in \mathbb R^{p},\quad \varepsilon \sim \mathcal N(0,\ \sigma^2 I_m)
  \end{equation*}
  \vspace{-2mm}
  \begin{tabular}{@{}ll@{}}
    $x$ & the $m$ recorded samples \\
    $z$ & the pitch curve, the \textbf{unknown} we are after \\
    $\Phi(z)$ & one column per harmonic wave; the curve fixes every column's \textbf{shape} \\
    $a$ & the harmonic amplitudes, the timbre; they only \textbf{scale} the columns \\
    $\varepsilon$ & what the harmonics do not explain
  \end{tabular}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{From the pitch curve to the design matrix}
  \small
  For any candidate curve, every harmonic's wave follows exactly:
  \vspace{1mm}
  \begin{alignat*}{2}
    &\text{pitch curve}\qquad & z &= \text{cents}(t)\\[1mm]
    &\text{frequency} & f_0(t) &= f_{\text{pitch}}\cdot 2^{\,\text{cents}(t)/1200}\\[1mm]
    &\text{phase} & \varphi_k(t) &= 2\pi k \int_0^{t} f_0,
       \qquad k = 1..K\\[1mm]
    &\text{design} & \Phi(z) &= \bigl[\cos\varphi_k(t),\ \sin\varphi_k(t)\bigr]_{k=1..K}
       \ \in \mathbb R^{m\times p}
  \end{alignat*}
  \vspace{1mm}

  Frequency is the speed, phase is the distance travelled: a small
  pitch error grows into a large phase error over time.
  \vspace{1mm}

  {\footnotesize The curve is held at $J$ knot values $u$ and
  interpolated to the $m$ samples; the columns are repeated per time
  chunk so loudness can change within the note, $p = 2KC$.}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Scoring a curve: the timbre integrated out}
  \small
  The amplitudes are unknown but \textbf{linear}. A Gaussian prior on
  them integrates out exactly:
  \begin{equation*}
    a \sim \mathcal N(0,\ \sigma_a^2 I_p)
    \quad\Longrightarrow\quad
    p(x \mid z) \;=\; \mathcal N\!\bigl(x;\ 0,\ \sigma_a^2\,\Phi(z)\Phi(z)^{\!\top} + \sigma^2 I_m\bigr)
  \end{equation*}
  \vspace{-1mm}
  \begin{itemize}
    \item One number per candidate curve, timbre-free. Every audio
          sample votes. No pitch tracker anywhere.
    \item Computed through the $p \times p$ matrix
          $G = \Phi^{\!\top}\Phi + \tfrac{\sigma^2}{\sigma_a^2} I_p$,
          never the $m \times m$ covariance: cost $O(mp^2)$, with
          $p \ll m$.
    \item $\sigma^2$ is the residual variance at the best amplitudes,
          fitted per note.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Finding the curve, with its uncertainty}
  \small
  \begin{alignat*}{2}
    &\text{prior on the curve} \qquad & u &\sim \mathcal N(0,\ K_{\text{knots}})\\[1mm]
    &\text{best curve} & \hat z &= \arg\max_z\ \bigl[\log p(x \mid z) + \log p(z)\bigr]\\[1mm]
    &\text{its band} & \Sigma_z &\approx
       \Bigl[-\nabla_z^2\bigl(\log p(x \mid z) + \log p(z)\bigr)\Bigr]^{-1}_{\hat z}
  \end{alignat*}
  \vspace{1mm}
  \begin{itemize}
    \item The score anchors $f_{\text{pitch}}$; $K_{\text{knots}}$ is the
          two-component GP of the previous chapter (vibrato band plus
          drift), so candidates are curves a player would produce.
    \item $\Phi(z)$ is differentiable in $z$: gradient descent through
          the synthesizer (cf.\ Engel et al.\ 2020), seconds per note.
    \item The band is the Laplace approximation: a sharply curved peak
          gives a narrow band.
  \end{itemize}
\end{frame}

\begin{frame}{Note boundaries: the chain model}
  \small
  A chain of $N$ consecutive notes, boundaries $b_1, \ldots, b_{N-1}$,
  one pitch curve $z_k$ per note. Each note is switched on and off by a
  soft window:
  \vspace{-1mm}
  \begin{alignat*}{2}
    &\text{window of note } k \qquad & w_k(t) &= \sigma\!\Bigl(\tfrac{t - b_{k-1}}{s}\Bigr)\,
       \sigma\!\Bigl(\tfrac{b_k - t}{s}\Bigr),
       \qquad \sigma(u) = \tfrac{1}{1 + e^{-u}},\quad s = 10\,\text{ms}\\[1mm]
    &\text{chain design} & \Phi(z, b) &= \bigl[\,w_1 \odot \Phi_1(z_1)\ \big|\ \cdots\ \big|\ w_N \odot \Phi_N(z_N)\,\bigr]
       \ \in \mathbb R^{m \times Np}\\[1mm]
    &\text{likelihood, as before} & p(x \mid z, b) &= \mathcal N\!\bigl(x;\ 0,\ \sigma_a^2\,\Phi\Phi^{\!\top} + \sigma^2 I_m\bigr)\\[1mm]
    &\text{start} & b^{(0)} &= \text{the score through the local tempo line (no annotation)}\\[1mm]
    &\text{joint fit} \qquad & (\hat z, \hat b) &= \arg\max_{z,\,b}\ \bigl[\log p(x \mid z, b) + \log p(z)\bigr],
       \qquad \text{bands: Laplace over } (z, b)
  \end{alignat*}
  \vspace{-2mm}
  \begin{itemize}
    \item $w_k$ fades note $k$ in after $b_{k-1}$, out at $b_k$ ($\odot$:
          columns scaled sample by sample). Differentiable in $b$: the
          boundaries move by the same gradient descent as the curves.
    \item The audio decides: a boundary settles where note $k$'s
          harmonics stop explaining the samples and note $k{+}1$'s start.
  \end{itemize}
\end{frame}
